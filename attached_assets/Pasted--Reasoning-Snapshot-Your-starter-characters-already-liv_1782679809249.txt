```js
// Module-level lock: prevents duplicate seed runs during React StrictMode,
// route transitions, refreshes, and background re-renders.
let seedPromise = null;

const SEED_RETRY_DELAY_MS = 1500;
const SEED_AUTH_TIMEOUT_MS = 30000;

function sleep(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

async function safeListCharacters(limit = 5000) {
  try {
    const list = await base44.entities.Character.list("-created_date", limit);
    return Array.isArray(list) ? list : [];
  } catch (err) {
    console.warn("[Anima] Could not list characters:", err?.message || err);
    return null;
  }
}

async function safeSeedStarterCharacters() {
  const starters = getStarterRoster();

  await waitForStoreAuth(SEED_AUTH_TIMEOUT_MS);

  const existing = await safeListCharacters(5000);

  // If the server cannot be reached, do not crash the app.
  // Let the UI load, then allow the next page visit/manual repair to try again.
  if (!existing) {
    console.warn("[Anima] Starter seed skipped because character store is unavailable.");
    return { added: 0, skipped: starters.length, unavailable: true };
  }

  const existingIds = new Set(existing.map((c) => c.id));
  const missing = starters.filter((c) => !existingIds.has(c.id));

  if (!missing.length) {
    return { added: 0, skipped: starters.length, unavailable: false };
  }

  const result = await upsertCharacters(missing);

  if (result.added) {
    console.log(`[Anima] Seeded ${result.added} starter character(s).`);
  }

  return {
    ...result,
    unavailable: false,
  };
}

async function doSeed() {
  try {
    return await safeSeedStarterCharacters();
  } catch (firstErr) {
    console.warn(
      "[Anima] Character seed first attempt failed, retrying once:",
      firstErr?.message || firstErr
    );

    try {
      await sleep(SEED_RETRY_DELAY_MS);
      return await safeSeedStarterCharacters();
    } catch (secondErr) {
      console.warn(
        "[Anima] Character seed failed after retry. App will continue:",
        secondErr?.message || secondErr
      );

      return {
        added: 0,
        skipped: getStarterRoster().length,
        unavailable: true,
        error: secondErr?.message || "Starter character seed failed.",
      };
    }
  }
}

export function seedCharactersIfNeeded() {
  if (!seedPromise) {
    seedPromise = doSeed()
      .then(async (result) => {
        // Photo backfill is cosmetic. Never let it block or break character loading.
        try {
          await backfillCharacterPhotos();
        } catch (err) {
          console.warn("[Anima] Photo backfill skipped:", err?.message || err);
        }

        return result;
      })
      .catch((err) => {
        // Final safety net. The starter seed must never white-screen the app.
        console.warn("[Anima] Background seed failed safely:", err?.message || err);
        seedPromise = null;

        return {
          added: 0,
          skipped: getStarterRoster().length,
          unavailable: true,
          error: err?.message || "Starter character seed failed.",
        };
      });
  }

  return seedPromise;
}
```
